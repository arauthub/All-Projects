Google Workspace Webhook Integration Guide

WorkspaceCurator Pro accepts push notifications and webhooks from every Google Workspace product through three standard methods:

Method 1: 1-Click Google Apps Script (Docs, Sheets & Forms)

  1. Open your Google Doc, Sheet, or Form.
  2. Click Extensions > Apps Script in the Google Workspace top menu.
  3. Delete any existing starter code in Code.gs and paste the script below.
  4. Set CURATOR_WEBHOOK_URL to your WorkspaceCurator endpoint (or tunnel URL).
  5. For Sheets/Forms: Click Triggers (alarm clock icon) > Add Trigger and select onEdit or onFormSubmit.
/**
 * WorkspaceCurator Pro - Google Apps Script Universal Trigger
 */
const CURATOR_WEBHOOK_URL = "http://localhost:8090/api/v1/webhook/google";

function onEdit(e) {
  if (!e) return;
  const sheet = SpreadsheetApp.getActiveSpreadsheet();
  const range = e.range;
  const payload = {
    product: "sheets",
    title: sheet.getName() + " (" + sheet.getActiveSheet().getName() + ")",
    sender: Session.getActiveUser().getEmail() || "Collaborator",
    content: "Cell edit at " + range.getA1Notation() + ": " + (e.value || "[cleared]"),
    url: sheet.getUrl()
  };
  UrlFetchApp.fetch(CURATOR_WEBHOOK_URL, {
    method: "post",
    contentType: "application/json",
    payload: JSON.stringify(payload)
  });
}

function onFormSubmit(e) {
  if (!e) return;
  const form = FormApp.getActiveForm();
  const responses = {};
  e.response.getItemResponses().forEach(item => {
    responses[item.getItem().getTitle()] = item.getResponse();
  });
  UrlFetchApp.fetch(CURATOR_WEBHOOK_URL, {
    method: "post",
    contentType: "application/json",
    payload: JSON.stringify({
      product: "forms",
      title: form ? form.getName() : "Form Response",
      sender: e.response.getRespondentEmail() || "Survey Respondent",
      responses: responses
    })
  });
}
